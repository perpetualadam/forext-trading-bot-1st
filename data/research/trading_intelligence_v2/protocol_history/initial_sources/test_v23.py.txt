class ImportTests(unittest.TestCase):
    def setUp(self):
        self.folder=TEMP/self.id().split('.')[-1]; self.folder.mkdir()
        self.manifest=rights(); files=[]
        for pair in PAIRS:
            path=self.folder/(pair+'.csv')
            price='150.000' if pair=='USD_JPY' else '1.10000'
            with path.open('w',encoding='utf-8',newline='') as f:
                w=csv.writer(f); w.writerow(SCHEMA)
                for minute in (0,5): w.writerow([f'2020-01-06T10:{minute:02}:00Z',pair,'true']+[price]*12)
            files.append(dict(instrument=pair,path=path.name,sha256=hashlib.sha256(path.read_bytes()).hexdigest(),
                              source_url='https://example.org/'+path.name,retrieved_utc='2026-10-27T00:00:00Z'))
        self.manifest['files']=files
    def alter(self,fn):
        path=self.folder/self.manifest['files'][0]['path']
        rows=list(csv.reader(path.open(newline=''))); fn(rows)
        with path.open('w',newline='') as f: csv.writer(f).writerows(rows)
        self.manifest['files'][0]['sha256']=hashlib.sha256(path.read_bytes()).hexdigest()
    def test_six_pair_import(self):
        data,report=import_dataset(self.manifest,self.folder)
        self.assertEqual(set(data),set(PAIRS)); self.assertEqual(set(report['independence'].values()),{'INDEPENDENT_UNTOUCHED'})
    def test_hash_mismatch(self):
        self.manifest['files'][0]['sha256']='0'*64
        with self.assertRaisesRegex(ResearchRejected,'hash_mismatch'): import_dataset(self.manifest,self.folder)
    def test_missing_rights(self):
        self.manifest['rights_reviewed']=False
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_naive_UTC_refused(self):
        self.alter(lambda r:r[1].__setitem__(0,'2020-01-06T10:00:00'))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_aware_offset_normalized(self):
        self.alter(lambda r:r[1].__setitem__(0,'2020-01-06T11:00:00+01:00'))
        self.assertEqual(import_dataset(self.manifest,self.folder)[0]['EUR_USD'][0].t,T)
    def test_duplicate(self):
        self.alter(lambda r:r.append(r[-1]))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_disorder_not_sorted(self):
        self.alter(lambda r:r.__setitem__(slice(1,None),list(reversed(r[1:]))))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_incomplete_not_discarded(self):
        self.alter(lambda r:r[1].__setitem__(2,'false'))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_crossed_market(self):
        self.alter(lambda r:r[1].__setitem__(7,'9.0'))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_identity(self):
        self.alter(lambda r:r[1].__setitem__(1,'USD_JPY'))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_no_synthetic_ask(self):
        self.alter(lambda r:r[0].__setitem__(-1,'missing_ask'))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_extra_fields_not_ignored(self):
        self.alter(lambda r:r[1].append('unexplained'))
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_missing_fields(self):
        self.alter(lambda r:r[1].pop())
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_multi_file_duplicate_refused(self):
        self.manifest['files'].append(dict(self.manifest['files'][0]))
        with self.assertRaisesRegex(ResearchRejected,'cross_file'): import_dataset(self.manifest,self.folder)
    def test_multiple_file_continuity(self):
        first=self.folder/self.manifest['files'][0]['path']; rows=list(csv.reader(first.open(newline='')))
        second=self.folder/'EUR_USD_part2.csv'
        with second.open('w',newline='') as f:
            w=csv.writer(f); w.writerow(rows[0]); w.writerow(['2020-01-06T10:10:00Z']+rows[-1][1:])
        self.manifest['files'].append(dict(instrument='EUR_USD',path=second.name,sha256=hashlib.sha256(second.read_bytes()).hexdigest(),
            source_url='https://example.org/'+second.name,retrieved_utc='2026-10-27T00:00:00Z'))
        data,_=import_dataset(self.manifest,self.folder); self.assertEqual(len(data['EUR_USD']),3)
    def test_ambiguity_of_timestamp_convention(self):
        self.manifest['timestamp_convention']='EST_UNSPECIFIED'
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_reproducible_manifest(self):
        self.assertEqual(import_dataset(self.manifest,self.folder),import_dataset(self.manifest,self.folder))
    def test_history_receipt_prevents_new_independence(self):
        with self.assertRaisesRegex(ResearchRejected,'independent_untouched'):
            import_dataset(self.manifest,self.folder,prior_intervals=((T,T+600),))
    def test_retrieval_before_completion_refused(self):
        self.manifest['files'][0]['retrieved_utc']='2019-01-01T00:00:00Z'
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)
    def test_broker_api_not_a_download_source(self):
        self.manifest['files'][0]['source_url']='https://api-fxtrade.oanda.com/v3/instruments/EUR_USD/candles'
        with self.assertRaises(ResearchRejected): import_dataset(self.manifest,self.folder)


class ProvenanceTests(unittest.TestCase):
    def test_actual_preregistered_protocol(self):

        spec=json.loads((package/'v23_experiment.json').read_text())

    def test_independent_period_boundary_is_half_open(self):
        self.assertEqual(classify(T,T+300,rights(),((T-300,T),)),'INDEPENDENT_UNTOUCHED')
    def test_previous(self): self.assertEqual(classify(T,T+300,dict(previously_examined=True)),'PREVIOUSLY_EXAMINED')
    def test_later_download_is_overlap(self):
        self.assertEqual(classify(timestamp('2026-09-01T00:00:00Z'),timestamp('2026-10-01T00:00:00Z'),rights()),'NEWLY_ACQUIRED_BUT_OVERLAPPING')
    def test_unknown_design_history(self):
        r=rights(); r['research_history_complete']=False
        self.assertEqual(classify(T,T+300,r),'PROVENANCE_UNVERIFIED')
    def test_frozen_spec_tamper(self):
        spec=dict(source_hashes={},holding={'M5':12}); h=canonical_digest(spec); spec['holding']['M5']=1
        with self.assertRaises(ResearchRejected): verify_spec(spec,h,TEMP)
    def test_frozen_source_tamper(self):

    def test_url_credentials_rejected(self):
        r=rights(); r['terms_url']='https://user:password@example.org/terms'

    def test_gaps_remain_gaps(self):
        b=[replace(fixtures.bar(0),t=T),replace(fixtures.bar(1),t=T+600)]
        q=quality('EUR_USD',b); self.assertEqual(q['gaps'][0]['missing_bars'],1); self.assertEqual(q['rows'],2)

        b=[replace(fixtures.bar(0),t=timestamp('2020-01-03T20:55:00Z')),replace(fixtures.bar(1),t=timestamp('2020-01-05T21:00:00Z'))]
        self.assertEqual(quality('EUR_USD',b)['gaps'][0]['kind'],'weekend_containing')
    def test_holiday_needs_documented_calendar(self):

class OverlapTests(unittest.TestCase):
    def test_joint_abstention_not_active_agreement(self):
        r=counts([None]*10,[None]*10)
        self.assertEqual(r['joint_abstention'],10); self.assertIsNone(r['conditional_direction_agreement'])
    def test_conflicts_and_one_only(self):
        r=counts(['BUY','BUY',None,'SELL'],['BUY','SELL','BUY',None])
        self.assertEqual((r['buy_agreement'],r['opposite_direction'],r['only_a'],r['only_b']),(1,1,1,1))
    def test_timestamps_must_align(self):
        with self.assertRaises(ValueError): counts(['BUY'],[])
    def test_duplicate_trade_ids(self):
        with self.assertRaises(ValueError): unique_trades([sample_trade(),sample_trade()])
    def test_paired_outcomes(self):
        a=[sample_trade(entry_time=T+i*300,net_r=float(i)) for i in range(3)]
        b=[dict(t,net_r=t['net_r']+1) for t in a]
        r=paired(a,b); self.assertEqual(r['matched'],3); self.assertAlmostEqual(r['outcome_R_correlation'],1)


class DiagnosticTests(unittest.TestCase):
    def test_gross_and_cost_drag(self):
        t=sample_trade(gross_R=-.2,turnover_account_usd=2200)
        self.assertEqual(cost_summary([t])['cost_class'],'NEGATIVE_GROSS_AND_COST_DRAG')
    def test_positive_gross_overwhelmed(self):
        t=sample_trade(gross_account=1,gross_R=.1,turnover_account_usd=2200)
        self.assertEqual(cost_summary([t])['cost_class'],'POSITIVE_GROSS_OVERWHELMED_BY_COSTS')
    def test_undefined_is_not_zero(self): self.assertIsNone(cost_summary([])['net_expectancy_R'])
    def test_components_reconcile(self):
        t=sample_trade(gross_R=-.2,turnover_account_usd=2200)
        s=cost_summary([t]); self.assertAlmostEqual(s['gross_net_difference_account'],s['spread_account']+s['slippage_account']+s['commission_account'])
    def test_intrabar_extreme_not_claimed_exact(self):
        b=[replace(fixtures.bar(0,high=1.101,low=1.099),t=T)]
        r=excursions(sample_trade(),b)
        self.assertEqual(r['excursion_status'],'OHLC_BOUNDS_EXIT_ORDER_UNKNOWN')
        self.assertAlmostEqual(r['MFE_lower_pips'],0); self.assertAlmostEqual(r['MFE_upper_pips'],9)
    def test_BUY_SELL_excursion_symmetry(self):
        b=[replace(fixtures.bar(0,high=1.101,low=1.099),t=T)]
        a=excursions(sample_trade(),b); c=excursions(sample_trade(direction='SELL',exit_price=1.1004),b)
        self.assertAlmostEqual(a['MFE_upper_pips'],c['MFE_upper_pips']); self.assertAlmostEqual(a['MAE_lower_pips'],c['MAE_lower_pips'])
    def test_gap_upper_unknown(self):
        b=[replace(fixtures.bar(0),t=T)]
        self.assertIsNone(excursions(sample_trade(exit_reason='DATA_GAP_CONSERVATIVE_STOP'),b)['MFE_upper_pips'])
    def test_JPY_pip(self): self.assertEqual(pip_size('USD_JPY'),.01)
    def test_conversion(self):
        q=Quotes(); q.put('USD_JPY',T,150,150.02); self.assertAlmostEqual(q.convert(-150,'JPY','USD',T),-1)
    def test_activation_observer_has_no_decision_effect(self):
        e=Existing(ROOT); observed=ProtectionObserver(e)
        a=SimpleNamespace(symbol='EUR_USD',open_time=T,last_bar=T-300,candidate={'strategy':'trend'},
            direction='BUY',entry_price=1.1,take_profit=1.102,max_profit_pips=15)
        b=deepcopy(a); x=e.protection(a,1.1009,atr_price=None); y=observed.protection(b,1.1009,atr_price=None)
        self.assertEqual(vars(a),vars(b)); self.assertEqual(x,y)
        event=observed.events[('EUR_USD',T,'trend')][0]
        self.assertTrue(event['active']); self.assertEqual(event['available_at'],T+300)
    def test_full_replay_observer_parity(self):
        data={'EUR_USD':[replace(fixtures.bar(i,price=1.1+i*.00001),t=T+i*300) for i in range(65)]}
        frames,packs,_=prepare(data); e=Existing(ROOT); window=(T+50*300,T+65*300)
        a=replay(data,{'EUR_USD':packs['EUR_USD']['M5']},e,window,protection=True,cost_breakdown=True)
        b=replay(data,{'EUR_USD':packs['EUR_USD']['M5']},ProtectionObserver(e),window,protection=True,cost_breakdown=True)
        self.assertEqual(a,b)
    def test_future_exit_extreme_only_changes_unknown_upper_bound(self):
        b=[replace(fixtures.bar(0,high=1.101,low=1.099),t=T)]
        c=[replace(b[0],h=1.2,bh=1.1999,ah=1.2001)]
        a=excursions(sample_trade(),b); z=excursions(sample_trade(),c)
        self.assertEqual(a['MFE_lower_pips'],z['MFE_lower_pips']); self.assertGreater(z['MFE_upper_pips'],a['MFE_upper_pips'])
    def test_scalp_trend_equality_counterexample(self):
        e=Existing(ROOT)
        p={'n':1,**{k:[v] for k,v in dict(close=1.1,ma_fast=1.1,ma_slow=1.,atr=.001,boll_up=1.2,boll_down=.9).items()}}
        self.assertEqual(e.signal('trend',0,p)[0],'BUY'); self.assertIsNone(e.signal('scalp',0,p)[0])
    def test_post_trade_not_selection_features(self):
        data={'EUR_USD':[replace(fixtures.bar(i),t=T+i*300) for i in range(60)]}; frames,packs,_=prepare(data)
        t=diagnose(sample_trade(), 'M5',data['EUR_USD'],frames['EUR_USD'],packs['EUR_USD'])
        self.assertIn('STOP_WITHIN_ONE_ENTRY_BAR',t['diagnostic_tags'])
        self.assertEqual(t['diagnostic_use'],'POST_TRADE_ONLY_NOT_SELECTION_FEATURES')
    def test_hold_exit_classification(self):
        data={'EUR_USD':[replace(fixtures.bar(i),t=T+i*300) for i in range(60)]}; f,p,_=prepare(data)
        t=diagnose(sample_trade(exit_reason='TIME_OR_SESSION_EXIT'),'M5',data['EUR_USD'],f['EUR_USD'],p['EUR_USD'])
        self.assertEqual(t['exit_diagnostic'],'HOLD_LIMIT_OR_FOLD_BOUNDARY')
    def test_coincident_time_exit_constraints_not_forced_to_one_cause(self):
        data={'EUR_USD':[replace(fixtures.bar(i),t=T+i*300) for i in range(60)]}; f,p,_=prepare(data)
        t=diagnose(sample_trade(exit_reason='TIME_OR_SESSION_EXIT'),'M5',data['EUR_USD'],f['EUR_USD'],p['EUR_USD'],
                   holding_limit_seconds=300,fold_end=T+300)
        self.assertEqual(t['exit_rule_triggers'],['HOLDING_TIMEOUT','FOLD_BOUNDARY_FLATTEN'])
